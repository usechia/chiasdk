Chia Widget Examples

Three ways to embed the Chia subscribe widget on your site. Replace pk_test_your_publishable_key with your actual publishable key from the Chia dashboard.

1. Inline Mode

The widget renders directly inside a container element.

Live Preview
<div id="chia-inline"></div>
<script src="https://cdn.jsdelivr.net/npm/@chiahq/widget/dist/chia-widget.min.js"></script>
<script>
  Chia.init({
    publishableKey: "pk_test_your_publishable_key",
    container: "#chia-inline",
    mode: "inline",
    planId: "your-plan-id",
    onSubscribed: function(result) {
      console.log("Subscribed:", result);
    },
    onError: function(error) {
      console.error("Error:", error);
    }
  });
</script>

2. Modal Mode

The widget opens as a centered overlay. Call widget.open() to show it.

Live Preview
<button id="open-modal">Subscribe</button>
<div id="chia-modal"></div>
<script>
  var modal = Chia.init({
    publishableKey: "pk_test_your_publishable_key",
    container: "#chia-modal",
    mode: "modal",
    onSubscribed: function(result) {
      console.log("Subscribed:", result);
    },
    onClose: function() {
      console.log("Modal closed");
    }
  });
  document.getElementById("open-modal").addEventListener("click", function() {
    modal.open();
  });
</script>

3. Button Mode

Combines a trigger button with the modal. Set mode: "button" and provide buttonText. Clicking the container opens the modal automatically.

Live Preview
<div id="chia-button"></div>
<script>
  Chia.init({
    publishableKey: "pk_test_your_publishable_key",
    container: "#chia-button",
    mode: "button",
    buttonText: "Subscribe Now",
    planSlug: "pro-monthly",
    theme: {
      primaryColor: "#7c3aed",
      borderRadius: "12px"
    },
    prefill: {
      phone: "+265999123456"
    },
    onSubscribed: function(result) {
      alert("Subscribed! ID: " + result.subscriberId);
    },
    onError: function(error) {
      alert("Error: " + error.message);
    },
    onClose: function() {
      console.log("Widget closed");
    },
    onReady: function() {
      console.log("Widget ready");
    }
  });
</script>

Configuration Reference

Chia.init({
  // Required
  publishableKey: "pk_test_...",    // Your publishable API key
  container: "#element" | element,  // CSS selector or DOM element

  // Display
  mode: "inline" | "modal" | "button",  // Default: "inline"
  buttonText: "Subscribe",              // Text for button mode trigger
  locale: "en",                         // Locale for formatting

  // Plan selection
  planId: "uuid",                // Show a specific plan by ID
  planSlug: "pro-monthly",       // Or by slug

  // Prefill
  prefill: {
    phone: "+265999123456",
    name: "Jane Doe"
  },

  // Theming
  theme: {
    primaryColor: "#2563eb",     // Brand color
    borderRadius: "8px",         // Corner radius
    fontFamily: "Inter, sans-serif"
  },

  // Callbacks
  onReady: function() {},
  onSubscribed: function(result) {},   // { id, subscriberId, paymentStatus, nextAction }
  onError: function(error) {},         // { message, code? }
  onClose: function() {},

  // Advanced
  apiBaseUrl: "https://api.usechia.com"  // Override API endpoint
});
